import time


# ==========================================
# 方法 1：直接使用次方運算
# ==========================================
def power2n_1(n):
    return 2 ** n


# ==========================================
# 方法 2a：遞迴
# power2n(n-1) + power2n(n-1)
# ==========================================
def power2n_2a(n):
    if n == 0:
        return 1

    return power2n_2a(n - 1) + power2n_2a(n - 1)


# ==========================================
# 方法 2b：遞迴
# 2 * power2n(n-1)
# ==========================================
def power2n_2b(n):
    if n == 0:
        return 1

    return 2 * power2n_2b(n - 1)


# ==========================================
# 方法 3：遞迴 + 查表
# ==========================================
memo = {0: 1}


def power2n_3(n):
    if n in memo:
        return memo[n]

    memo[n] = power2n_3(n - 1) + power2n_3(n - 1)

    return memo[n]


# ==========================================
# 測試函數
# ==========================================
def test_function(func, n):
    start = time.perf_counter()

    result = func(n)

    end = time.perf_counter()

    elapsed = end - start

    return result, elapsed


# ==========================================
# 主程式
# ==========================================

n = 100

print("=" * 60)
print(f"測試 n = {n}")
print("=" * 60)


# ------------------------------------------
# 方法 1
# ------------------------------------------
result1, time1 = test_function(power2n_1, n)

print("\n方法 1：2 ** n")
print("結果：", result1)
print("執行時間：", time1, "秒")


# ------------------------------------------
# 方法 2b
# ------------------------------------------
result2b, time2b = test_function(power2n_2b, n)

print("\n方法 2b：2 * power2n(n-1)")
print("結果：", result2b)
print("執行時間：", time2b, "秒")


# ------------------------------------------
# 方法 3
# ------------------------------------------
memo.clear()
memo[0] = 1

result3, time3 = test_function(power2n_3, n)

print("\n方法 3：遞迴 + 查表")
print("結果：", result3)
print("執行時間：", time3, "秒")


# ------------------------------------------
# 方法 2a
# ------------------------------------------
# 方法 2a 在 n = 100 時會產生非常大量的遞迴呼叫，
# 因此不直接執行 n = 100。
#
# 改用 n = 20 來測試。
# ------------------------------------------

small_n = 20

result2a, time2a = test_function(power2n_2a, small_n)

print("\n方法 2a：power2n(n-1) + power2n(n-1)")
print(f"測試 n = {small_n}")
print("結果：", result2a)
print("執行時間：", time2a, "秒")

print("\n" + "=" * 60)
print("方法 2a 的 n = 100 不直接執行")
print("=" * 60)

print("""
方法 2a 每次遞迴都會產生兩個新的遞迴呼叫。

時間複雜度約為 O(2^n)。

當 n = 100 時，
函數呼叫次數約為：

2^(101) - 1

約等於 2.54 × 10^30 次。

因此 n = 100 時可以視為「出不來」。
""")
